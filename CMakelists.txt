cmake_minimum_required(VERSION 3.15)
project(tdxpy_formulas VERSION 0.1.0 LANGUAGES CXX)

# ============================================================================
# 项目全局配置
# ============================================================================

# 设置默认构建类型
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Release" CACHE STRING "构建类型 (Debug/Release)" FORCE)
endif()

# 设置C++标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 设置输出目录
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# ============================================================================
# 架构选择配置
# ============================================================================

# 允许用户选择架构（默认为32位以兼容通达信）
set(TDXPY_TARGET_ARCHITECTURE "Win32" CACHE STRING "目标架构 (Win32/x64)")
set_property(CACHE TDXPY_TARGET_ARCHITECTURE PROPERTY STRINGS "Win32" "x64")

# 如果通过 -A 选项设置了平台，自动调整架构
if(CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    # 如果 CMake 指定了 x64 平台，强制使用 x64 架构
    set(TDXPY_TARGET_ARCHITECTURE "x64" CACHE STRING "目标架构 (Win32/x64)" FORCE)
endif()

if(TDXPY_TARGET_ARCHITECTURE STREQUAL "Win32")
    # 32位构建
    set(CMAKE_GENERATOR_PLATFORM "Win32")
    set(TDXPY_TARGET_ARCH "x86")
    message(STATUS "目标架构: 32位 (Win32)")
elseif(TDXPY_TARGET_ARCHITECTURE STREQUAL "x64")
    # 64位构建
    set(CMAKE_GENERATOR_PLATFORM "x64")
    set(TDXPY_TARGET_ARCH "x64")
    message(STATUS "目标架构: 64位 (x64)")
endif()

# ============================================================================
# Windows特定配置
# ============================================================================

if(WIN32)
    # Windows版本定义
    add_definitions(-DWIN32 -D_WINDOWS -D_UNICODE -DUNICODE)
    
    # DLL导出控制
    if(MSVC)
        # 设置编译选项
        set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} /MT /O2 /Ob2 /D NDEBUG")
        set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} /MTd /Zi /Ob0 /Od /RTC1")
        
        # DLL导出宏
        add_definitions(-DTDXPY_FORMULAS_EXPORTS)
        
        # 警告级别
        if(CMAKE_CXX_FLAGS MATCHES "/W[0-4]")
            string(REGEX REPLACE "/W[0-4]" "/W3" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
        else()
            set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /W3")
        endif()
        
        # SDL检查
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /sdl")
        
        # 一致性模式
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /permissive-")
    endif()
endif()

# ============================================================================
# Python配置 - 根据架构选择对应版本的Python
# ============================================================================

# 允许用户自定义Python路径
set(PYTHON_ROOT_32 "${CMAKE_SOURCE_DIR}/third_party/Python3142-32" CACHE PATH "32位Python路径")
set(PYTHON_ROOT_64 "${CMAKE_SOURCE_DIR}/third_party/Python3142-64" CACHE PATH "64位Python路径")

# 根据架构选择Python路径
if(TDXPY_TARGET_ARCHITECTURE STREQUAL "Win32")
    # 32位构建：使用32位Python
    set(PYTHON_ROOT "${PYTHON_ROOT_32}")
    message(STATUS "Python版本: 3.14.2 (32位)")
elseif(TDXPY_TARGET_ARCHITECTURE STREQUAL "x64")
    # 64位构建：使用64位Python
    set(PYTHON_ROOT "${PYTHON_ROOT_64}")
    message(STATUS "Python版本: 3.14.2 (64位)")
else()
    message(FATAL_ERROR "未知的目标架构: ${TDXPY_TARGET_ARCHITECTURE}")
endif()

set(PYTHON_INCLUDE_DIR "${PYTHON_ROOT}/include")

# 根据构建类型选择Python库和DLL
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    # Debug版本使用python314_d.lib 和 python314_d.dll
    set(PYTHON_LIBRARY_NAME "python314_d.lib")
    set(PYTHON_DLL_NAME "python314_d.dll")
else()
    # Release版本使用python314.lib 和 python314.dll
    set(PYTHON_LIBRARY_NAME "python314.lib")
    set(PYTHON_DLL_NAME "python314.dll")
endif()

set(PYTHON_LIBRARY "${PYTHON_ROOT}/libs/${PYTHON_LIBRARY_NAME}")
set(PYTHON_DLL_PATH "${PYTHON_ROOT}/${PYTHON_DLL_NAME}")

# 检查Python依赖
if(NOT EXISTS "${PYTHON_ROOT}")
    message(FATAL_ERROR "Python安装目录不存在: ${PYTHON_ROOT}
请确保Python 3.14.2 ${TDXPY_TARGET_ARCH}位已安装到指定目录")
endif()

if(NOT EXISTS "${PYTHON_INCLUDE_DIR}/Python.h")
    message(FATAL_ERROR "Python头文件未找到: ${PYTHON_INCLUDE_DIR}/Python.h
请检查Python安装是否完整")
endif()

if(NOT EXISTS "${PYTHON_LIBRARY}")
    message(FATAL_ERROR "Python库文件未找到: ${PYTHON_LIBRARY}")
endif()

if(NOT EXISTS "${PYTHON_DLL_PATH}")
    message(FATAL_ERROR "Python DLL未找到: ${PYTHON_DLL_PATH}")
endif()

message(STATUS "Python路径: ${PYTHON_ROOT}")
message(STATUS "Python包含目录: ${PYTHON_INCLUDE_DIR}")
message(STATUS "Python库: ${PYTHON_LIBRARY}")
message(STATUS "Python DLL: ${PYTHON_DLL_PATH}")

# ============================================================================
# JSON库配置
# ============================================================================

set(JSON_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/include/json")
set(JSON_LIBS_BASE_DIR "${CMAKE_SOURCE_DIR}/Libs/json")

# 检查JSON包含目录
if(NOT EXISTS "${JSON_INCLUDE_DIR}")
    message(FATAL_ERROR "JSON库头文件目录不存在: ${JSON_INCLUDE_DIR}")
endif()

# 根据目标架构选择JSON库目录
if(TDXPY_TARGET_ARCHITECTURE STREQUAL "Win32")
    # 32位构建：使用win32目录
    set(JSON_ARCH_DIR "${JSON_LIBS_BASE_DIR}/win32")
elseif(TDXPY_TARGET_ARCHITECTURE STREQUAL "x64")
    # 64位构建：使用x64目录
    set(JSON_ARCH_DIR "${JSON_LIBS_BASE_DIR}/x64")
else()
    message(FATAL_ERROR "未知的目标架构: ${TDXPY_TARGET_ARCHITECTURE}")
endif()

# 根据构建类型选择Debug或Release版本
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(JSON_LIB_DIR "${JSON_ARCH_DIR}/Debug")
    set(JSON_LIBRARY_NAME "jsoncpp.lib")
    set(JSON_DLL_NAME "jsoncpp.dll")
else()
    set(JSON_LIB_DIR "${JSON_ARCH_DIR}/Release")
    set(JSON_LIBRARY_NAME "jsoncpp.lib")
    set(JSON_DLL_NAME "jsoncpp.dll")
endif()

# 检查JSON库目录是否存在
if(NOT EXISTS "${JSON_LIB_DIR}")
    message(FATAL_ERROR "JSON库目录不存在: ${JSON_LIB_DIR}
请确保${TDXPY_TARGET_ARCHITECTURE} ${CMAKE_BUILD_TYPE}版本的JSON库已放置在正确位置:
  期望路径: ${JSON_LIB_DIR}
  基础路径: ${JSON_LIBS_BASE_DIR}
  项目结构应为:
    Libs/json/
      ├── win32/    (32位)
      │   ├── Debug/
      │   │   ├── jsoncpp.lib
      │   │   └── jsoncpp.dll
      │   └── Release/
      │       ├── jsoncpp.lib
      │       └── jsoncpp.dll
      └── x64/      (64位)
          ├── Debug/
          │   ├── jsoncpp.lib
          │   └── jsoncpp.dll
          └── Release/
              ├── jsoncpp.lib
              └── jsoncpp.dll")
endif()

# 查找JSON库文件
find_file(JSON_LIBRARY
    NAMES ${JSON_LIBRARY_NAME}
    PATHS ${JSON_LIB_DIR}
    NO_DEFAULT_PATH
    REQUIRED
)

find_file(JSON_DLL_PATH
    NAMES ${JSON_DLL_NAME}
    PATHS ${JSON_LIB_DIR}
    NO_DEFAULT_PATH
    REQUIRED
)

if(NOT JSON_LIBRARY)
    message(FATAL_ERROR "JSON库未找到: ${JSON_LIB_DIR}/${JSON_LIBRARY_NAME}")
endif()

if(NOT JSON_DLL_PATH)
    message(FATAL_ERROR "JSON DLL未找到: ${JSON_LIB_DIR}/${JSON_DLL_NAME}")
endif()

message(STATUS "JSON包含目录: ${JSON_INCLUDE_DIR}")
message(STATUS "JSON架构目录: ${JSON_ARCH_DIR}")
message(STATUS "JSON库目录: ${JSON_LIB_DIR}")
message(STATUS "JSON库: ${JSON_LIBRARY}")
message(STATUS "JSON DLL: ${JSON_DLL_PATH}")

# 源文件定义

# 项目主源文件
set(TDXPY_SOURCES
    ${CMAKE_SOURCE_DIR}/src/dll_main.cpp

    ${CMAKE_SOURCE_DIR}/src/tdxpy_plugin_registry.cpp

    ${CMAKE_SOURCE_DIR}/src/tdxpy_logger.cpp
    ${CMAKE_SOURCE_DIR}/src/tdxpy_env_var.cpp
)

set(TDXPY_INCLUDE_DIRS
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/src
    ${PYTHON_INCLUDE_DIR}
    ${JSON_INCLUDE_DIR}
)

# ============================================================================
# 测试配置
# ============================================================================

option(BUILD_TESTING "构建测试" OFF)

if(BUILD_TESTING)
    # 设置测试源文件目录
    set(TEST_SOURCE_DIR "${CMAKE_SOURCE_DIR}/tests/cpp_tests")
    
    # 检查测试源文件目录是否存在
    if(NOT EXISTS "${TEST_SOURCE_DIR}")
        message(WARNING "测试源文件目录不存在: ${TEST_SOURCE_DIR}")
        set(TEST_SOURCES "")
    else()
        # 添加测试源文件
        file(GLOB TEST_SOURCES 
            "${TEST_SOURCE_DIR}/*.cpp"
        )
        
        # 获取测试源文件数量
        list(LENGTH TEST_SOURCES TEST_SOURCES_COUNT)
        
        # 检查是否有测试源文件
        if(TEST_SOURCES_COUNT EQUAL 0)
            message(WARNING "在目录 ${TEST_SOURCE_DIR} 中未找到测试源文件")
        else()
            # 创建测试目标
            add_executable(tdxpy_formulas_tests ${TEST_SOURCES})
            
            # 设置测试目标属性
            set_target_properties(tdxpy_formulas_tests PROPERTIES
                OUTPUT_NAME "tdxpy_formulas_tests"
                RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests"
            )
            
            # 设置包含目录
            target_include_directories(tdxpy_formulas_tests PRIVATE 
                ${TDXPY_INCLUDE_DIRS}
                ${TEST_SOURCE_DIR}
                gtest gtest_main  # 链接 gtest_main（如果使用 Google Test 的主程序）
            )
            
            target_sources(tdxpy_formulas_tests PRIVATE ${TDXPY_SOURCES})

            # 链接库
            target_link_libraries(tdxpy_formulas_tests
                    ${PYTHON_LIBRARY}  # Python库
                    ${JSON_LIBRARY}    # JSON库
                )
            
            # Windows特定链接
            if(WIN32)
                target_link_options(tdxpy_formulas_tests PRIVATE
                    /NODEFAULTLIB:library
                )
                
                # 在Windows上链接ODBC库
                target_link_libraries(tdxpy_formulas_tests
                    odbc32
                    odbccp32
                )
                
                # 链接gtest库（需要确保gtest库已安装）
                # 如果使用Google Test作为源码，需要包含gtest_main
                find_package(GTest QUIET)
                if(GTest_FOUND)
                    target_link_libraries(tdxpy_formulas_tests GTest::gtest GTest::gtest_main GTest::gmock GTest::gmock_main)
                else()
                    # 如果未找到预安装的gtest，尝试添加本地gtest
                    if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/googletest")
                        add_subdirectory(${CMAKE_SOURCE_DIR}/third_party/googletest)
                        target_link_libraries(tdxpy_formulas_tests gtest gtest_main gmock gmock_main)
                    else()
                        message(WARNING "未找到Google Test库，测试将无法链接")
                    endif()
                endif()
            endif()
            
            # 启用测试
            enable_testing()
            
            # 添加测试
            add_test(NAME tdxpy_formulas_tests
                COMMAND tdxpy_formulas_tests
                WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/tests
            )
            
            message(STATUS "测试目标已添加: tdxpy_formulas_tests")
            message(STATUS "测试源文件目录: ${TEST_SOURCE_DIR}")
            message(STATUS "找到的测试源文件数量: ${TEST_SOURCES_COUNT}")
        endif()
    endif()
endif()

# 创建部署目标
add_custom_target(deploy
    # 创建部署目录
    COMMAND ${CMAKE_COMMAND} -E make_directory
        "${CMAKE_SOURCE_DIR}/deploy/tdxpy_formulas_${TDXPY_TARGET_ARCH}_${CMAKE_BUILD_TYPE}"
    
    # 复制主DLL
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:tdxpy_formulas>"
        "${CMAKE_SOURCE_DIR}/deploy/tdxpy_formulas_${TDXPY_TARGET_ARCH}_${CMAKE_BUILD_TYPE}/"
    
    # 复制Python DLL
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${PYTHON_DLL_PATH}"
        "${CMAKE_SOURCE_DIR}/deploy/tdxpy_formulas_${TDXPY_TARGET_ARCH}_${CMAKE_BUILD_TYPE}/"
    
    # 复制JSON DLL
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${JSON_DLL_PATH}"
        "${CMAKE_SOURCE_DIR}/deploy/tdxpy_formulas_${TDXPY_TARGET_ARCH}_${CMAKE_BUILD_TYPE}/"
    
    # 复制Python安装目录
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${PYTHON_ROOT}"
        "${CMAKE_SOURCE_DIR}/deploy/tdxpy_formulas_${TDXPY_TARGET_ARCH}_${CMAKE_BUILD_TYPE}/${PYTHON_DEPLOY_DIR}"
    
    DEPENDS tdxpy_formulas
    COMMENT "创建部署包"
)
